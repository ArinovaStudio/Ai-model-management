generator client {
  provider             = "prisma-client-py"
  recursive_type_depth = -1
}

datasource db {
  provider     = "postgresql"
  url          = env("DATABASE_URL")
  relationMode = "prisma"
}

model Project {
  id        String   @id @default(uuid())
  name      String
  summary   String?
  status    String   @default("active")
  createdAt DateTime @default(now())
  updatedAt DateTime @default(now()) @updatedAt

  projectInfo     ProjectInfo?
  planDocs        PlanDoc[]
  docs            Doc[]
  featureRequests FeatureRequest[]
  technologies    ProjectTechnology[]
  riskBlockages   RiskBlockage[]
}

model ProjectInfo {
  id          String   @id @default(uuid())
  projectId   String   @unique
  project     Project  @relation(fields: [projectId], references: [id], onDelete: Cascade)
  name        String   @default("Unnamed Project")
  description String?
  startDate   DateTime
  deadline    DateTime
  scope       String?
  constraints String?
  metadata    String?
  createdAt   DateTime @default(now())
  updatedAt   DateTime @default(now()) @updatedAt
}

model PlanDoc {
  id        String   @id @default(uuid())
  projectId String
  project   Project  @relation(fields: [projectId], references: [id], onDelete: Cascade)
  title     String
  content   String
  embedding String?
  metadata  String?
  createdAt DateTime @default(now())
  updatedAt DateTime @default(now()) @updatedAt
}

model Doc {
  id        String   @id @default(uuid())
  projectId String
  project   Project  @relation(fields: [projectId], references: [id], onDelete: Cascade)
  title     String
  content   String
  docType   String?
  embedding String?
  metadata  String?
  createdAt DateTime @default(now())
  updatedAt DateTime @default(now()) @updatedAt
}

model FeatureRequest {
  id                  String   @id @default(uuid())
  projectId           String
  project             Project  @relation(fields: [projectId], references: [id], onDelete: Cascade)
  title               String
  description         String
  priority            String   @default("medium")
  status              String   @default("proposed")
  category            String?
  sourceDocId         String?
  sourceDocType       String?
  embedding           String?
  isAmbiguous         Boolean  @default(false)
  clarificationPrompt String?
  metadata            String?
  createdAt           DateTime @default(now())
  updatedAt           DateTime @default(now()) @updatedAt
}

model ProjectTechnology {
  id           String   @id @default(uuid())
  projectId    String
  project      Project  @relation(fields: [projectId], references: [id], onDelete: Cascade)
  name         String   @default("Unknown Tech")
  category     String   @default("backend")
  version      String?
  isDetected   Boolean  @default(true)
  dependencies String?
  metadata     String?
  createdAt    DateTime @default(now())
  updatedAt    DateTime @default(now()) @updatedAt
}

model RiskBlockage {
  id              String   @id @default(uuid())
  projectId       String
  project         Project  @relation(fields: [projectId], references: [id], onDelete: Cascade)
  type            String
  category        String
  title           String
  description     String
  severity        String   @default("medium")
  status          String   @default("open")
  sourceFeatureId String?
  sourceDocId     String?
  reportType      String   @default("risk")
  metadata        String?
  createdAt       DateTime @default(now())
  updatedAt       DateTime @default(now()) @updatedAt
}
